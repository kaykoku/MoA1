  <script>
    (function () {
      const ROOMS = ["메인홀", "룸2", "룸3", "룸4", "룸5", "룸6"];
      const HOURS = Array.from({ length: 15 }, (_, idx) => idx + 9);
      const STORAGE_BOOKINGS = "room_pwa_bookings_v1";
      const STORAGE_PIN = "room_pwa_admin_pin_v1";
      const STORAGE_UNLOCKED = "room_pwa_admin_unlocked_v1";

      let selectedIndices = [];
      let isAdminUnlocked = sessionStorage.getItem(STORAGE_UNLOCKED) === "true";

      function formatHourSlot(h) {
        const pad = (n) => String(n).padStart(2, "0");
        return `${pad(h)}:00 ~ ${pad(h + 1)}:00`;
      }

      function getTodayStr() {
        const now = new Date();
        const y = now.getFullYear();
        const m = String(now.getMonth() + 1).padStart(2, "0");
        const d = String(now.getDate()).padStart(2, "0");
        return `${y}-${m}-${d}`;
      }

      function showToast(msg) {
        const el = document.getElementById("toast");
        el.textContent = msg;
        el.classList.add("show");
        clearTimeout(showToast._timer);
        showToast._timer = setTimeout(() => el.classList.remove("show"), 2600);
      }

      function loadBookings() {
        try {
          const raw = localStorage.getItem(STORAGE_BOOKINGS);
          if (raw) return JSON.parse(raw);
        } catch (e) {}
        const today = getTodayStr();
        const initial = [
          {
            id: "seed-1",
            room: "메인홀",
            date: today,
            slotIndices: [1, 2],
            slots: [formatHourSlot(10), formatHourSlot(11)],
            name: "김민수",
            org: "대전창업모임",
            phone: "010-2222-3333",
            email: "minsu@example.com",
            title: "주간 네트워킹",
            attendees: "30명",
            purpose: "멤버 교류회",
            isPublic: "오픈",
            equip: ["빔프로젝터", "음향/마이크"],
            note: "10분 전 입장 예정",
            status: "APPROVED",
            createdAt: new Date().toISOString()
          }
        ];
        saveBookings(initial);
        return initial;
      }

      function saveBookings(list) {
        localStorage.setItem(STORAGE_BOOKINGS, JSON.stringify(list));
      }

      function getAdminPin() {
        return localStorage.getItem(STORAGE_PIN) || "1234";
      }

      function setAdminPin(newPin) {
        localStorage.setItem(STORAGE_PIN, newPin);
      }

      function isSlotBlocked(bookings, room, date, slotIndex, excludeId = null) {
        return bookings.some(
          (b) =>
            b.id !== excludeId &&
            b.room === room &&
            b.date === date &&
            (b.status === "PENDING" || b.status === "APPROVED") &&
            Array.isArray(b.slotIndices) &&
            b.slotIndices.includes(slotIndex)
        );
      }

      function isConsecutive(arr) {
        if (!arr.length) return true;
        const sorted = [...arr].sort((a, b) => a - b);
        for (let i = 1; i < sorted.length; i++) {
          if (sorted[i] !== sorted[i - 1] + 1) return false;
        }
        return true;
      }

      function renderSlots() {
        const room = document.getElementById("room-select").value;
        const date = document.getElementById("date-input").value;
        const grid = document.getElementById("slot-grid");
        const bookings = loadBookings();

        selectedIndices = selectedIndices.filter((idx) => !isSlotBlocked(bookings, room, date, idx));
        if (!isConsecutive(selectedIndices)) {
          selectedIndices = [];
        }

        grid.innerHTML = "";
        HOURS.forEach((h, idx) => {
          const btn = document.createElement("button");
          btn.type = "button";
          btn.className = "slot-btn";
          btn.textContent = formatHourSlot(h);
          const blocked = isSlotBlocked(bookings, room, date, idx);
          const isSel = selectedIndices.includes(idx);

          if (blocked) btn.disabled = true;
          if (isSel) btn.classList.add("selected");

          btn.addEventListener("click", () => {
            if (blocked) return;
            if (selectedIndices.includes(idx)) {
              const candidate = selectedIndices.filter((item) => item !== idx);
              if (!isConsecutive(candidate)) {
                showToast("중간 시간만 빼면 연속 시간이 끊깁니다. 양끝부터 해제하세요.");
                return;
              }
              selectedIndices = candidate;
            } else {
              const candidate = [...selectedIndices, idx].sort((a, b) => a - b);
              if (!isConsecutive(candidate)) {
                showToast("연속된 시간대만 선택할 수 있습니다.");
                return;
              }
              selectedIndices = candidate;
            }
            renderSlots();
          });

          grid.appendChild(btn);
        });

        const summary = document.getElementById("selected-slot-text");
        if (!selectedIndices.length) {
          summary.textContent = "선택된 시간: 없음 (1시간 이상 연속 선택)";
        } else {
          const sorted = [...selectedIndices].sort((a, b) => a - b);
          const startStr = formatHourSlot(HOURS[sorted[0]]).split(" ~ ")[0];
          const endStr = formatHourSlot(HOURS[sorted[sorted.length - 1]]).split(" ~ ")[1];
          summary.textContent = `선택된 시간: ${startStr} ~ ${endStr} (${sorted.length}시간 연속)`;
        }
      }

      function syncAdminView() {
        const lockScreen = document.getElementById("admin-lock-screen");
        const content = document.getElementById("admin-content");
        if (isAdminUnlocked) {
          lockScreen.style.display = "none";
          content.style.display = "block";
          renderAdmin();
        } else {
          lockScreen.style.display = "block";
          content.style.display = "none";
          updatePinDots("");
          document.getElementById("pin-input").value = "";
        }
      }

      function updatePinDots(val) {
        const dots = document.querySelectorAll("#pin-dots .pin-dot");
        dots.forEach((dot, i) => {
          dot.classList.toggle("filled", i < val.length);
        });
      }

      function attemptUnlock() {
        const input = document.getElementById("pin-input");
        const entered = input.value.trim();
        if (entered === getAdminPin()) {
          isAdminUnlocked = true;
          sessionStorage.setItem(STORAGE_UNLOCKED, "true");
          showToast("관리자 대시보드가 열렸습니다.");
          syncAdminView();
        } else {
          showToast("비밀번호(PIN)가 일치하지 않습니다.");
          input.value = "";
          updatePinDots("");
          input.focus();
        }
      }

      function renderAdmin() {
        const bookings = loadBookings();
        const targetDate = document.getElementById("admin-date-filter").value || getTodayStr();
        const statusFilter = document.getElementById("status-filter").value;

        document.getElementById("stat-total").textContent = bookings.length;
        document.getElementById("stat-pending").textContent = bookings.filter((b) => b.status === "PENDING").length;
        document.getElementById("stat-approved").textContent = bookings.filter((b) => b.status === "APPROVED").length;
        document.getElementById("stat-rejected").textContent = bookings.filter((b) => b.status === "REJECTED").length;

        const table = document.getElementById("timetable");
        let thead = "<thead><tr><th>시간</th>" + ROOMS.map((r) => `<th>${r}</th>`).join("") + "</tr></thead>";
        let tbody = "<tbody>";
        HOURS.forEach((h, idx) => {
          tbody += `<tr><td><strong>${formatHourSlot(h)}</strong></td>`;
          ROOMS.forEach((room) => {
            const hit = bookings.find(
              (b) =>
                b.room === room &&
                b.date === targetDate &&
                (b.status === "PENDING" || b.status === "APPROVED") &&
                Array.isArray(b.slotIndices) &&
                b.slotIndices.includes(idx)
            );
            if (!hit) {
              tbody += `<td class="cell-free">가능</td>`;
            } else if (hit.status === "APPROVED") {
              tbody += `<td class="cell-approved">[완료] ${hit.title}</td>`;
            } else {
              tbody += `<td class="cell-pending">[대기] ${hit.title}</td>`;
            }
          });
          tbody += "</tr>";
        });
        tbody += "</tbody>";
        table.innerHTML = thead + tbody;

        const listEl = document.getElementById("booking-list");
        const filtered = bookings.filter((b) => statusFilter === "ALL" || b.status === statusFilter);

        if (!filtered.length) {
          listEl.innerHTML = `<p class="helper">표시할 예약 신청 내역이 없습니다.</p>`;
          return;
        }

        const statusBadge = (st) => {
          if (st === "APPROVED") return `<span class="badge badge-approved">승인 완료</span>`;
          if (st === "REJECTED") return `<span class="badge badge-rejected">반려/취소</span>`;
          return `<span class="badge badge-pending">승인 대기</span>`;
        };

        listEl.innerHTML = filtered
          .map((b) => {
            const sorted = [...(b.slotIndices || [])].sort((a, c) => a - c);
            const timeLabel = sorted.length
              ? `${formatHourSlot(HOURS[sorted[0]]).split(" ~ ")[0]} ~ ${formatHourSlot(HOURS[sorted[sorted.length - 1]]).split(" ~ ")[1]}`
              : (b.slots || []).join(", ");

            return `
              <div class="booking-item">
                <div class="booking-top">
                  <div>
                    <div class="booking-title">${escapeHtml(b.title)} · <span style="color:var(--primary);">${escapeHtml(b.room)}</span></div>
                    <div class="booking-meta">
                      <span>📅 ${escapeHtml(b.date)}</span>
                      <span>⏰ ${escapeHtml(timeLabel)}</span>
                      <span>👤 ${escapeHtml(b.name)} (${escapeHtml(b.org)})</span>
                      <span>📞 ${escapeHtml(b.phone)}</span>
                      <span>✉️ ${escapeHtml(b.email)}</span>
                      <span>👥 ${escapeHtml(b.attendees)}</span>
                      <span>🔓 ${escapeHtml(b.isPublic)}</span>
                    </div>
                  </div>
                  ${statusBadge(b.status)}
                </div>
                <div class="helper" style="color:#334155;">
                  <strong>목적:</strong> ${escapeHtml(b.purpose)}
                  ${b.equip && b.equip.length ? ` · <strong>장비:</strong> ${escapeHtml(b.equip.join(", "))}` : ""}
                  ${b.note ? ` · <strong>비고:</strong> ${escapeHtml(b.note)}` : ""}
                </div>
                <div class="booking-actions">
                  ${b.status !== "APPROVED" ? `<button type="button" class="btn btn-success" data-action="approve" data-id="${b.id}">✓ 승인 완료</button>` : ""}
                  ${b.status !== "REJECTED" ? `<button type="button" class="btn btn-danger" data-action="reject" data-id="${b.id}">✕ 반려/취소</button>` : ""}
                  <button type="button" class="btn btn-outline" data-action="delete" data-id="${b.id}">삭제</button>
                </div>
              </div>
            `;
          })
          .join("");
      }

      function escapeHtml(str) {
        return String(str || "")
          .replace(/&/g, "&amp;")
          .replace(/</g, "&lt;")
          .replace(/>/g, "&gt;")
          .replace(/"/g, "&quot;");
      }

      document.querySelectorAll(".nav-btn").forEach((btn) => {
        btn.addEventListener("click", () => {
          const tab = btn.dataset.tab;
          document.querySelectorAll(".nav-btn").forEach((b) => b.classList.toggle("active", b === btn));
          document.getElementById("view-request").classList.toggle("active", tab === "request");
          document.getElementById("view-admin").classList.toggle("active", tab === "admin");
          if (tab === "request") {
            renderSlots();
          } else {
            syncAdminView();
          }
        });
      });

      document.getElementById("room-select").addEventListener("change", () => {
        selectedIndices = [];
        renderSlots();
      });

      document.getElementById("date-input").addEventListener("change", () => {
        selectedIndices = [];
        renderSlots();
      });

      document.getElementById("clear-slots-btn").addEventListener("click", () => {
        selectedIndices = [];
        renderSlots();
      });

      document.getElementById("reservation-form").addEventListener("submit", (e) => {
        e.preventDefault();
        const room = document.getElementById("room-select").value;
        const date = document.getElementById("date-input").value;

        if (!selectedIndices.length) {
          showToast("이용할 시간대를 1시간 이상 선택해 주세요.");
          return;
        }
        if (!isConsecutive(selectedIndices)) {
          showToast("연속된 시간대만 예약할 수 있습니다.");
          return;
        }

        const bookings = loadBookings();
        const hasConflict = selectedIndices.some((idx) => isSlotBlocked(bookings, room, date, idx));
        if (hasConflict) {
          showToast("선택한 시간 중 이미 예약된 구간이 있습니다.");
          renderSlots();
          return;
        }

        const sorted = [...selectedIndices].sort((a, b) => a - b);
        const equip = Array.from(document.querySelectorAll('input[name="equip"]:checked')).map((el) => el.value);
        const isPublic = document.querySelector('input[name="isPublic"]:checked').value;

        const newItem = {
          id: "bk-" + Date.now(),
          room,
          date,
          slotIndices: sorted,
          slots: sorted.map((idx) => formatHourSlot(HOURS[idx])),
          name: document.getElementById("applicant-name").value.trim(),
          org: document.getElementById("applicant-org").value.trim(),
          phone: document.getElementById("applicant-phone").value.trim(),
          email: document.getElementById("applicant-email").value.trim(),
          title: document.getElementById("event-title").value.trim(),
          attendees: document.getElementById("attendees").value.trim(),
          purpose: document.getElementById("event-purpose").value.trim(),
          isPublic,
          equip,
          note: document.getElementById("extra-note").value.trim(),
          status: "PENDING",
          createdAt: new Date().toISOString()
        };

        bookings.unshift(newItem);
        saveBookings(bookings);
        selectedIndices = [];
        e.target.reset();
        document.getElementById("room-select").value = room;
        document.getElementById("date-input").value = date;
        renderSlots();
        showToast("예약 신청이 접수되었습니다. (승인 대기)");
      });

      const pinInput = document.getElementById("pin-input");
      pinInput.addEventListener("input", () => updatePinDots(pinInput.value));
      pinInput.addEventListener("keydown", (e) => {
        if (e.key === "Enter") {
          e.preventDefault();
          attemptUnlock();
        }
      });

      document.getElementById("pin-unlock-btn").addEventListener("click", attemptUnlock);

      document.querySelectorAll(".key-btn").forEach((btn) => {
        btn.addEventListener("click", () => {
          const k = btn.dataset.key;
          if (k === "clear") {
            pinInput.value = "";
            updatePinDots("");
          } else if (k === "ok") {
            attemptUnlock();
          } else if (pinInput.value.length < 8) {
            pinInput.value += k;
            updatePinDots(pinInput.value);
          }
        });
      });

      document.getElementById("lock-now-btn").addEventListener("click", () => {
        isAdminUnlocked = false;
        sessionStorage.removeItem(STORAGE_UNLOCKED);
        syncAdminView();
        showToast("관리자 화면이 잠겼습니다.");
      });

      document.getElementById("change-pin-btn").addEventListener("click", () => {
        const nextPin = prompt("새 관리자 PIN(숫자 4~8자리)을 입력하세요:");
        if (nextPin === null) return;
        const trimmed = nextPin.trim();
        if (!/^\d{4,8}$/.test(trimmed)) {
          showToast("숫자 4~8자리로 입력해 주세요.");
          return;
        }
        setAdminPin(trimmed);
        showToast("관리자 PIN이 변경되었습니다.");
      });

      document.getElementById("admin-date-filter").addEventListener("change", renderAdmin);
      document.getElementById("status-filter").addEventListener("change", renderAdmin);

      document.getElementById("booking-list").addEventListener("click", (e) => {
        const btn = e.target.closest("button[data-action]");
        if (!btn) return;
        const id = btn.dataset.id;
        const action = btn.dataset.action;
        let bookings = loadBookings();
        const target = bookings.find((b) => b.id === id);
        if (!target) return;

        if (action === "approve") {
          const conflict = bookings.some(
            (other) =>
              other.id !== target.id &&
              other.room === target.room &&
              other.date === target.date &&
              other.status === "APPROVED" &&
              other.slotIndices.some((idx) => target.slotIndices.includes(idx))
          );
          if (conflict) {
            showToast("이미 승인 완료된 일정과 시간이 겹쳐 승인할 수 없습니다.");
            return;
          }
          target.status = "APPROVED";
          showToast("예약이 승인 완료 처리되었습니다.");
        } else if (action === "reject") {
          target.status = "REJECTED";
          showToast("예약이 반려 처리되어 해당 시간대가 열렸습니다.");
        } else if (action === "delete") {
          if (!confirm("이 예약 내역을 완전히 삭제할까요?")) return;
          bookings = bookings.filter((b) => b.id !== id);
          showToast("예약 내역이 삭제되었습니다.");
        }

        saveBookings(bookings);
        renderAdmin();
        renderSlots();
      });

      if ("https:" === location.protocol) {
        const manifestObj = {
          name: "대관 예약 관리 시스템",
          short_name: "대관예약",
          start_url: ".",
          display: "standalone",
          background_color: "#f1f5f9",
          theme_color: "#1e3a8a"
        };
        const manifestBlob = new Blob([JSON.stringify(manifestObj)], { type: "application/manifest+json" });
        const manifestUrl = URL.createObjectURL(manifestBlob);
        const link = document.createElement("link");
        link.rel = "manifest";
        link.href = manifestUrl;
        document.head.appendChild(link);

        if ("serviceWorker" in navigator) {
          const swCode = `
            self.addEventListener("install", (e) => self.skipWaiting());
            self.addEventListener("activate", (e) => e.waitUntil(self.clients.claim()));
            self.addEventListener("fetch", (e) => {
              e.respondWith(fetch(e.request).catch(() => caches.match(e.request)));
            });
          `;
          const swBlob = new Blob([swCode], { type: "application/javascript" });
          const swUrl = URL.createObjectURL(swBlob);
          navigator.serviceWorker.register(swUrl).catch(() => {});
        }
      }

      const today = getTodayStr();
      document.getElementById("date-input").value = today;
      document.getElementById("admin-date-filter").value = today;
      renderSlots();
    })();
